public protocol Named {
  var name: String { get }
}

public protocol Counted {
  var count: Int { get }
}

public type NamedCounted = Named & Counted

public func describe<T>(_ x: T) -> String where T: Named & Counted {
  return "\(x.name): \(x.count)"
}

public struct Card: ViewModifier {
  var padded: Bool = true
  func body(content: Content) -> some View {
    content.padding(.md)
  }
}

@Transparent
public struct UserID {
  public let raw: String
}

public func fail(_ message: String) -> Never {
  fatal(message)
}

public func useCounter(start: Int) -> Int {
  @State var n = start
  return n
}

public func firstNegative(_ grid: [[Int]]) -> Int? {
  var found: Int? = none
  outer: for row in grid {
    for x in row {
      if x > 9 { continue outer }
      if x < 0 {
        found = x
        break outer
      }
    }
  }
  while true { break }
  return found
}

public final class Session {
  public init() {}
}

public func same(_ a: Session, _ b: Session?) -> Bool {
  return a === b || a !== b
}

public func isDigits(_ s: String) throws -> Bool {
  let digits = try Regex("^\\d+$")
  return s.wholeMatch(of: digits) != none
}

public func closures() -> Int {
  let rows = { () -> [Int] in [] }
  let twice = { (a: Int) -> Int in a * 2 }
  return twice(rows().count)
}

public enum StoreError: Error {
  case closed
}

actor Ledger {
  fileprivate static var opened = 0
  internal private(set) var total = 0
  private func add(_ n: Int) {
    self.total += n
  }
}

extension Session: Named {
  public var name: String { "session" }
}

public func run(_ f: () throws -> Int) rethrows -> Int {
  return try f()
}

public func label(_ x: any Named, _ n: Int) throws -> String {
  guard n >= 0 else { throw StoreError.closed }
  switch n {
  case 0:
    return x.name
  default:
    break
  }
  do {
    return "\(try run { n })"
  } catch {
    return "failed"
  }
}
