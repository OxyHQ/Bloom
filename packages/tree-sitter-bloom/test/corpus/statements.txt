fn area(s: Shape) -> Float {
  match s {
    case .circle(r) => 3.14159 * r * r
    case .rect(w, h) if w > 0 => { w * h }
    case Shape.dot => 0.0
    case _ => draft = Draft()
  }
}

  @State var n = start
  outer: for row in grid {
    for x in row {
      if x > 1 { continue outer }
      if x < 0 { break outer }
    }
  }
  return n
