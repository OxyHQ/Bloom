pub view Composer(maxLength: Int = 500, onSend: async (Draft) -> Result<Unit, String>) {
  let { t } = useTranslation()
  @state draft = Draft()
  @state sending: Bool = false
  @env scheme = Env.colorScheme

  let left = maxLength - draft.text.count

  async fn send() {
    sending = true
  }

  effect(id: draft.text) {
    defer { sending = false }
  }

  equal(old, new) { old.maxLength == new.maxLength }

  body {
    VStack {
      TextField($draft.text, t("compose.placeholder"))
      Text("${left}").foreground(left < 0 ? "error" : "muted")
    }
    .spacing("md")
    .padding("lg")
  }
}

public struct PostRow: View, Equatable {
  let post: Post
  var dense: Bool = false
  let onLike: () -> Void
  @Binding var isOn: Bool
  @State private var draft = ""
  @Environment(\.locale) private var locale
  @Query(Post.self, sort: \.createdAt, order: .reverse) var posts: [Post]

  func send() { draft = "" }

  var body: some View {
    HStack { Text(post.text) }
      .padding(.md)
      .foreground(.primary.opacity(0.1))
      .onAppear { send() }
  }
}

view Feed(posts: [Post]) {
  body {
    VStack {
      if posts.isEmpty {
        Text("Nothing yet")
      } else {
        for post in posts {
          PostRow(post: post).id(post.id)
        }
      }
      Button { refresh() } label: { Text("Refresh") }
    }
  }
}
